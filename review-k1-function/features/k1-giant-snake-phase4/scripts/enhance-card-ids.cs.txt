var types = System.AppDomain.CurrentDomain.GetAssemblies().SelectMany(a => { try { return a.GetTypes(); } catch { return new System.Type[0]; } }).ToList();
var pmT = types.First(x => x.FullName == "DeepUI.PopupManager"); var pm = pmT.GetProperty("I", bf).GetValue(null);
var ecT = types.First(x => x.FullName == "UI.UIEnhanceCard");
