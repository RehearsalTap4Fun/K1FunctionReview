var types = System.AppDomain.CurrentDomain.GetAssemblies().SelectMany(a => { try { return a.GetTypes(); } catch { return new System.Type[0]; } }).ToList();
var bf = System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.FlattenHierarchy;
var worldT = types.First(x => x.Name == "World" && x.GetField("LogicMgr") != null);

var api = types.First(x => x.Name == "SystemAPI" && x.GetMethods().Any(m => m.Name == "QueryEntities"));
var q = api.GetMethods().First(m => m.Name == "QueryEntities" && m.IsGenericMethodDefinition && m.GetGenericArguments().Length == 1 && m.GetParameters().Length == 1);

  if ((bool)ct.GetField("IsDead").GetValue(u) || (bool)ct.GetField("ReachedDefense").GetValue(u)) continue; k++;

  var w = gw.Invoke(run, new object[]{ id }); var wt = w.GetType();

  restore.Invoke(w, new object[]{ np, x, y, false });

return System.DateTime.Now.ToString("HH:mm:ss") + " " + string.Join(" | ", outp);
